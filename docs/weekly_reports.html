# Set weekly target per team
weekly_target <- 500

# Aggregate data by week and team
weekly_performance <- aggregate_visits %>%
  # Ensure date is in Date format
  mutate(date = as.Date(date)) %>%
  # Create week variable (adjust based on your week definition)
  mutate(
    week = floor_date(date, unit = "week", week_start = 1),  # Starts on Sunday
    # Alternative: week = floor_date(date, unit = "week", week_start = 1) # Starts on Monday
    week_label = format(week, "%b %d, %Y")) %>%
  # Group by team and week
  group_by(team, week) %>%
  summarise(achieved = sum(n, na.rm = TRUE),
    .groups = "drop") %>%
  # Add target and calculate variance
  mutate(target = weekly_target,
    variance = achieved - target,
    achievement_pct = round((achieved / target) * 100, 1),
    status = case_when(
      achievement_pct >= 100 ~ "✅ Met/Exceeded",
      achievement_pct >= 80 ~ "⚠️ Near Target",
      TRUE ~ "❌ Below Target" )) %>%
  # Sort by team and week
  arrange(team, week)

# First, create the team_service_performance dataframe
team_service_performance <- performance_serv %>%
  # Rename to match target dataset
  rename(service_name = service_col) %>%
  
  # Join with targets
  left_join(
    nzatonse_team_service_targets,
    by = c("team" = "team", "service_name" = "service_col")
  ) %>%
  
  # Calculate metrics
  mutate(
    # For a single week (Feb 09-13, 2026), weekly_target is the target
    weekly_target = weekly_target,
    expected_total = weekly_target,  # Since it's just one week
    achievement_pct = (total_clients / weekly_target) * 100,
    
    # Create status based on achievement percentage
    status = case_when(
      achievement_pct >= 100 ~ "✅ Met/Exceeded",
      achievement_pct >= 80 ~ "⚠️ Near Target",
      TRUE ~ "❌ Below Target"
    ),
    
    gap_to_target = total_clients - weekly_target,
    
    # Weekly average is just total_clients since it's one week
    weekly_average = total_clients
  ) %>%
  
  # Select and arrange columns
  select(team, service_name, achieved = total_clients, weekly_average, 
         weekly_target, expected_total, achievement_pct, status, gap_to_target) %>%
  arrange(team, desc(achievement_pct))
